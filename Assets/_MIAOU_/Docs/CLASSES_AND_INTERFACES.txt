  PARTIE I — INTERFACES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  NOTE: UdonSharp ne supporte pas les interfaces au runtime.
  Elles servent de CONTRAT DE CONCEPTION : définir l'API publique attendue.
  Chaque classe les implémente manuellement (mêmes propriétés/méthodes).

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IEntity                                                                 │
  │ Fichier : Core/Interfaces/IEntity.cs                                    │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Toute entité identifiable du jeu (joueur, NPC, véhicule, objet…)        │
  │ Propriétés : EntityId, EntityName, Type (EntityType), IsActive          │
  │ Implémenté par : HealthController (via EntityId/EntityName/Type)        │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IDamageable                                                             │
  │ Fichier : Core/Interfaces/IDamageable.cs                                │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Tout ce qui peut recevoir des dégâts et mourir                          │
  │ Propriétés : Health, MaxHealth, IsAlive, CanTakeDamage                  │
  │ Méthodes   : TakeDamage(damage, attackerId, DamageType)                 │
  │              Heal(amount)   Die(killerId)                               │
  │ Implémenté par : HealthController                                       │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IHittable                                                               │
  │ Fichier : Core/Interfaces/IHittable.cs  (à créer)                       │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Point d'entrée physique des dégâts (collider) avant le HealthController │
  │ Déclenché par un raycast ou une collision de projectile                 │
  │ Méthodes  : OnHit(rawDamage, attackerId, DamageType, hitPoint, normal)  │
  │ Retourne  : float damageAfterModifiers (transmis à HealthController)    │
  │ Implémenté par : HitZone                                                │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ ITeamMember                                                             │
  │ Fichier : Core/Interfaces/ITeamMember.cs                                │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Toute entité appartenant à une équipe (joueur, NPC, véhicule)           │
  │ Propriétés : TeamIndex                                                  │
  │ Méthodes   : SetTeam(teamIndex)                                         │
  │ Implémenté par : PlayerController, NpcController, VehicleController     │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IWeapon                                                                 │
  │ Fichier : Core/Interfaces/IWeapon.cs                                    │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Toute arme capable de tirer et de gérer ses munitions                   │
  │ Propriétés : WeaponName, WeaponType, CurrentAmmo, MaxAmmo,              │
  │              ReserveAmmo, Damage, FireRate, Range, IsEquipped, CanFire  │
  │ Méthodes   : Fire()  Reload()  Equip()  Unequip()                       │
  │ Implémenté par : WeaponProp                                             │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IInteractable                                                           │
  │ Fichier : Core/Interfaces/IInteractable.cs                              │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Tout objet avec lequel un joueur peut interagir (E / VRC_Interactable)  │
  │ Propriétés : CanInteract, InteractionText, InteractionRange             │
  │ Méthodes   : OnInteract(player)  OnLookAt(player)  OnLookAway(player)   │
  │ Implémenté par : InteractProp                                           │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IArmored                                                                │
  │ Fichier : Core/Interfaces/IArmored.cs  (à créer)                        │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Entité possédant un système d'armure absorbant les dégâts               │
  │ Propriétés : ArmorPoints, MaxArmorPoints                                │
  │ Méthodes   : AbsorbDamage(damage, DamageType, zoneId) : float           │
  │              RepairArmor(amount)                                        │
  │ Implémenté par : ArmorSystem                                            │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IBuffable                                                               │
  │ Fichier : Core/Interfaces/IBuffable.cs  (à créer)                       │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Entité pouvant recevoir des buffs/debuffs temporaires                   │
  │ Méthodes   : ApplyBuff(buffType, magnitude, duration)                   │
  │              RemoveBuff(buffType)                                       │
  │              HasBuff(buffType) : bool                                   │
  │              GetStatMultiplier(statType) : float                        │
  │ Implémenté par : BuffSystem                                             │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IInventoryHolder                                                        │
  │ Fichier : Core/Interfaces/IInventoryHolder.cs  (à créer)                │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Entité possédant un inventaire de slots d'armes/objets                  │
  │ Méthodes   : TryAddItem(weaponProp) : bool                              │
  │              RemoveItem(slotIndex)                                      │
  │              GetActiveWeapon() : WeaponProp                             │
  │              SwitchSlot(slotIndex)                                      │
  │ Implémenté par : InventorySystem                                        │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IGameMode                                                               │
  │ Fichier : Core/Interfaces/IGameMode.cs  (à créer)                       │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Contrat commun à tous les game modes, lu par le Manager et RoundManager │
  │ Propriétés : ModeName, CurrentState (GameState), TimeLeft, ScoreLimit   │
  │ Méthodes   : StartGame()  EndGame()  ResetGame()                        │
  │              GetWinnerTeam() : int                                      │
  │ Implémenté par : GameMode (base), tous les modes spécifiques            │
  └─────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ IVehicleOccupant                                                        │
  │ Fichier : Core/Interfaces/IVehicleOccupant.cs  (à créer)                │
  ├─────────────────────────────────────────────────────────────────────────┤
  │ Entité pouvant entrer/sortir d'un véhicule                              │
  │ Méthodes   : OnEnterVehicle(vehicle)   OnExitVehicle(vehicle)           │
  │ Implémenté par : PlayerController, NpcController                        │
  └─────────────────────────────────────────────────────────────────────────┘

  PARTIE II — CLASSES (hiérarchie complète)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  UdonSharpBehaviour  (VRChat SDK)
  └── M2922_Base
        │  Gère : Log/Debug, référence au Manager, OnValidate éditeur
        │  Note : Classe racine de TOUS les scripts M2922
        │
        ├── M2922_Manager
        │     Gère : Référence centrale (EventBus, TeamManager, SpawnManager…)
        │            Point d'entrée unique pour tous les modules
        │     Interfaces : —
        │
        ├── M2922_EventBus
        │     Gère : Pub/sub d'événements, pool de EventData, sync réseau
        │     Interfaces : —
        │
        ├── M2922_EventData
        │     Gère : Conteneur de données d'un événement (slot du pool)
        │     Interfaces : —
        │
        ├── ─── COMBAT ──────────────────────────────────────────────────────
        │
        ├── M2922_HealthController
        │     Gère : HP (actuel/max), TakeDamage, Die, Heal, Respawn, regen,
        │            invincibilité, effets visuels mort/dégât, sync réseau
        │     Interfaces : → IEntity  → IDamageable
        │     Liens : →| EventBus (OnPlayerDamaged, OnPlayerKilled, OnPlayerHealed)
        │
        ├── M2922_HitZone
        │     Gère : Collider d'une zone anatomique, multiplicateur de dégâts,
        │            transmission vers HealthController après calcul
        │     Interfaces : → IHittable
        │     Liens : ↔ HealthController (parent)
        │              ↔ ArmorSystem (même GO ou parent)
        │
        ├── M2922_ArmorSystem
        │     Gère : Points d'armure par zone, absorption avant HealthController,
        │            dégradation des points d'armure, réparation
        │     Interfaces : → IArmored
        │     Liens : ↔ HealthController (même GO)
        │
        ├── M2922_ResistanceTable
        │     Gère : Tableau de résistances par DamageType (0..1),
        │            réduction appliquée après ArmorSystem
        │     Interfaces : —
        │     Liens : ↔ HealthController (même GO, consulté dans TakeDamage)
        │
        ├── M2922_BuffSystem
        │     Gère : Buffs/debuffs actifs (type, magnitude, durée), expiration,
        │            exposition de multiplicateurs pour HP/vitesse/résistances
        │     Interfaces : → IBuffable
        │     Liens : ↔ HealthController (modificateur regen/max HP)
        │              →| EventBus (OnBuffApplied, OnBuffExpired)
        │
        ├── ─── PROPS ───────────────────────────────────────────────────────
        │
        ├── M2922_Prop
        │     Gère : Cycle vie/mort/respawn visuel d'un objet de map,
        │            colliders, VFX mort/respawn, reset de transform
        │     Interfaces : —
        │     Liens : ↔ HealthController (optionnel, même GO)
        │     Note : CLASSE DE BASE de tous les objets de map
        │
        │     ├── M2922_WeaponProp
        │     │     Gère : Tir (raycast), ammo, rechargement, VRCPickup,
        │     │            retour à l'origine après drop, désactivation si détruite
        │     │     Interfaces : → IWeapon
        │     │     Liens : ↔ HealthController (cible, via GetComponentInParent)
        │     │              →| EventBus (OnWeaponFired, OnWeaponEquipped…)
        │     │
        │     ├── M2922_InteractProp
        │     │     Gère : Interaction joueur (VRC_Interactable), états de l'objet,
        │     │            logique conditionnelle (équipe, état de partie)
        │     │     Interfaces : → IInteractable
        │     │     Liens : →| EventBus (OnObjectInteracted)
        │     │              ← EventBus (OnGameStateChanged pour activer/désactiver)
        │     │
        │     ├── M2922_CollectableProp
        │     │     Gère : Disparition au ramassage, effet immédiat (heal/ammo/buff),
        │     │            respawn après délai
        │     │     Interfaces : → IInteractable
        │     │     Liens : ↔ HealthController (cible, pour Heal)
        │     │              ↔ BuffSystem (cible, pour ApplyBuff)
        │     │              →| EventBus (OnItemPickedUp)
        │     │
        │     └── M2922_StructureProp
        │           Gère : Structure destructible multi-états (intact/endommagé/détruit),
        │                  blocage ou ouverture de passage selon état
        │           Interfaces : —
        │           Liens : ↔ HealthController (même GO)
        │                    →| EventBus (OnStructureDestroyed)
        │
        ├── M2922_Projectile
        │     Gère : Physique du projectile (Rigidbody), dégâts à l'impact,
        │            rayon d'explosion, durée de vie, VFX,
        │            transmission de l'attackerId depuis l'arme source
        │     Interfaces : —
        │     Liens : ↔ HitZone / HealthController (cible à l'impact)
        │              →| EventBus (OnProjectileImpact)
        │
        ├── ─── JOUEUR ──────────────────────────────────────────────────────
        │
        ├── M2922_PlayerController
        │     Gère : Appartenance à une équipe, respawn au bon point,
        │            team visuals, statistiques (kills/deaths/score/assists),
        │            état (Alive/Dead/Spectating/InVehicle),
        │            abonnement aux événements de partie
        │     Interfaces : → ITeamMember  → IVehicleOccupant
        │     Liens : ↔ HealthController (même GO)
        │              ↔ TeamManager     (référence assignée)
        │              ↔ SpawnManager    (référence assignée)
        │              ↔ InventorySystem (même GO, optionnel)
        │              ↔ BuffSystem      (même GO, optionnel)
        │              ← EventBus (OnPlayerKilled, OnGameStarted…)
        │
        ├── M2922_InventorySystem
        │     Gère : Slots d'armes (primaire/secondaire/mêlée/grenade),
        │            slots d'équipement (armure, casque),
        │            ajout/retrait/swap d'objets, drop à la mort
        │     Interfaces : → IInventoryHolder
        │     Liens : ↔ WeaponProp (référence dans les slots)
        │              ↔ ArmorSystem (application de l'armure équipée)
        │              →| EventBus (OnItemPickedUp, OnItemDropped)
        │              ← EventBus (OnPlayerDied → drop tous les objets)
        │
        ├── ─── NPC ─────────────────────────────────────────────────────────
        │
        ├── M2922_NpcController
        │     Gère : IA comportementale (Idle/Patrol/Alert/Chase/Attack/Dead),
        │            équipe, détection des ennemis, attaque, loot à la mort
        │     Interfaces : → ITeamMember  → IVehicleOccupant
        │     Liens : ↔ HealthController (même GO)
        │              ↔ TeamManager     (vérification équipe avant attaque)
        │              →| EventBus (OnNpcKilled, OnNpcAlert)
        │              ← EventBus (OnGameStateChanged pour activer/désactiver)
        │
        ├── ─── VÉHICULE ────────────────────────────────────────────────────
        │
        ├── M2922_VehicleController        ⊂ M2922_Prop
        │     Gère : Sièges (VRCStation), physique moteur/direction,
        │            armement monté, transmission de dégâts aux occupants,
        │            éjection des occupants à la mort, ownership VRChat
        │     Interfaces : → ITeamMember
        │     Liens : ↔ HealthController (même GO)
        │              ↔ TeamManager     (équipe du véhicule)
        │              →| EventBus (OnVehicleDestroyed, OnPlayerEnterVehicle)
        │              ← EventBus (OnRoundEnd → reset/destroy)
        │
        ├── ─── ÉQUIPES ─────────────────────────────────────────────────────
        │
        ├── M2922_TeamManager
        │     Gère : Définition des équipes (nom/couleur/actif),
        │            attribution des joueurs, auto-balance,
        │            vérification friendly fire, scores par équipe,
        │            compteur de joueurs par équipe
        │     Interfaces : —
        │     Liens : →| EventBus (OnTeamScoreChanged, OnTeamChanged)
        │              ← EventBus (OnPlayerKilled → peut incrémenter score)
        │
        ├── ─── SPAWN ───────────────────────────────────────────────────────
        │
        ├── M2922_SpawnManager
        │     Gère : Sélection du point de spawn selon équipe, distance,
        │            mode (aléatoire, séquentiel, le plus éloigné),
        │            spawn initial et respawn en cours de partie
        │     Interfaces : —
        │     Liens : ↔ SpawnPoint[] (référence directe)
        │              ← EventBus (OnGameStarted → unlock respawn)
        │
        ├── M2922_SpawnPoint
        │     Gère : Position/rotation de spawn, tag d'équipe, type de spawn,
        │            disponibilité (occupé ou libre)
        │     Interfaces : —
        │     Liens : ↔ SpawnManager (appartient à sa liste)
        │
        ├── ─── TRIGGERS & ZONES ────────────────────────────────────────────
        │
        ├── M2922_ZoneTrigger
        │     Gère : Détection entrée/sortie de joueurs dans une zone,
        │            dégâts continus (DOT), soins continus,
        │            capture de zone (timer, équipe dominante),
        │            buff/debuff de zone
        │     Interfaces : —
        │     Liens : ↔ HealthController (cibles : TakeDamage / Heal)
        │              ↔ TeamManager     (équipe dominante pour capture)
        │              →| EventBus (OnZoneCaptured, OnZoneContested)
        │              ← EventBus (OnRoundStart → reset état de zone)
        │
        ├── ─── GAME MODE ────────────────────────────────────────────────────
        │
        ├── M2922_GameMode  (base abstraite)
        │     Gère : État de partie (Waiting/Preparing/InProgress/Finished),
        │            timer synced, conditions de victoire, reset de map
        │     Interfaces : → IGameMode
        │     Liens : →| EventBus (OnGameStateChanged, OnGameFinished)
        │              ← EventBus (OnPlayerKilled, OnZoneCaptured…)
        │
        │     ├── M2922_GameMode_TDM
        │     │     Gère : Score limit, kills par équipe, vainqueur par score
        │     │
        │     ├── M2922_GameMode_FFA
        │     │     Gère : Score par joueur individuel, vainqueur unique
        │     │
        │     ├── M2922_GameMode_Domination
        │     │     Gère : Zones à capturer, score continu par zone tenue
        │     │
        │     ├── M2922_GameMode_SearchDestroy
        │     │     Gère : Bombe, vies limitées, pas de respawn mid-round
        │     │
        │     └── M2922_GameMode_Horde
        │           Gère : Vagues de NPC, difficulté croissante,
        │                  survie collective des joueurs
        │
        ├── M2922_RoundManager
        │     Gère : Succession des manches, scoreboard inter-manches,
        │            meilleur sur N manches, statistiques fin de partie
        │     Interfaces : —
        │     Liens : ↔ GameMode (référence directe)
        │              ← EventBus (OnGameFinished → lancer la manche suivante)
        │              →| EventBus (OnRoundStart, OnMatchFinished)
        │
        ├── ─── SCORING ─────────────────────────────────────────────────────
        │
        ├── M2922_ScoringSystem
        │     Gère : Calcul et attribution des points (kill, assist, capture,
        │            destruction véhicule…), multiplicateurs killstreak/headshot,
        │            classement joueurs, transmission au GameMode
        │     Interfaces : —
        │     Liens : ← EventBus (OnPlayerKilled, OnZoneCaptured, OnVehicleDestroyed…)
        │              ↔ TeamManager (AddTeamScore)
        │              ↔ GameMode    (vérifier si score limit atteint)
        │
        ├── M2922_KillstreakSystem
        │     Gère : Compteur de kills consécutifs par joueur,
        │            attribution de récompenses échelonnées (UAV, frappe…),
        │            reset à chaque mort
        │     Interfaces : —
        │     Liens : ← EventBus (OnPlayerKilled, OnPlayerDied)
        │              →| EventBus (OnKillstreakReward)
        │
        └── ─── HUD & FEEDBACK ──────────────────────────────────────────────

              M2922_KillFeed
                    Gère : Affichage des kills récents (attaquant → victime + arme),
                           annonces headshot, killstreak, objectif capturé
                    Liens : ← EventBus (OnPlayerKilled, OnZoneCaptured…)

              M2922_HudController
                    Gère : Health bar, ammo display, timer, score affiché,
                           indicateur dégâts reçus (direction), minimap
                    Liens : ↔ HealthController (lecture Health pour HP bar)
                             ↔ WeaponProp      (CurrentAmmo / ReserveAmmo)
                             ← EventBus (OnPlayerDamaged → flash directionnel)
